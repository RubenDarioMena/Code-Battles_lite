1 bucle n0 | for (const x of [1, 2]) { if (x === 1) continue a; console.log(x); }
    etiqueta: Por cada [x] de [[1, 2]]
    lectura:  Repite las filas (2) a (4) una vez por cada [x] de [[1, 2]]. Después, sigue en (5).
    sigue: sig=5 si=2 no=· salida=fin
2 decision n1 | if (x === 1) continue a;
    etiqueta: ¿[x] vale 1?
    lectura:  Si [x] vale 1, sigue en (3); si no, salta a (4).
    sigue: sig=4 si=3 no=· salida=fila:3
3 paso (lateral) n2 | continue a;
    etiqueta: continue a
    lectura:  Se salta el resto de la vuelta y vuelve a (1).
    sigue: sig=1 si=· no=·
4 paso n1 | console.log(x);
    etiqueta: Imprime [x]
    lectura:  Escribe [x] en la consola. Luego vuelve a (1).
    sigue: sig=1 si=· no=·
5 paso n0 | console.log("fin");
    etiqueta: Imprime "fin"
    lectura:  Escribe "fin" en la consola. Aquí acaba el programa.
    sigue: sig=· si=· no=·
rieles: 4→1 3→1
consola: 2 / fin
resumen: Eso es todo el programa.
pasos:
  1 Entra en el bucle.
  2 Calcula [x === 1]…
  3 continue: se salta el resto de la vuelta y vuelve a (1).
  4 Escribe «"undefined"» en la consola.
  5 Escribe «"fin"» en la consola.
